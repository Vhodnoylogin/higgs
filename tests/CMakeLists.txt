cmake_minimum_required(VERSION 3.20)
project(HiggsHandServiceTests LANGUAGES C CXX)
enable_testing()

include(../cmake/HiggsServices.cmake)
add_library(higgs_hand_services STATIC ${HIGGS_SERVICE_SOURCES})
target_include_directories(higgs_hand_services PUBLIC "${HIGGS_SERVICE_INCLUDE_DIR}")
target_compile_features(higgs_hand_services PUBLIC cxx_std_20)
find_package(Threads REQUIRED)
target_link_libraries(higgs_hand_services PUBLIC Threads::Threads)

add_executable(hand_broker hand_broker_tests.cpp hand_c_consumer.c)
target_compile_features(hand_broker PRIVATE c_std_11)
add_executable(hand_authority hand_authority_tests.cpp)
add_executable(native_regions ../src/test/native_regions_tests.cpp)
foreach(target IN ITEMS hand_broker hand_authority native_regions)
    target_link_libraries(${target} PRIVATE higgs_hand_services)
    add_test(NAME ${target} COMMAND ${target})
endforeach()
